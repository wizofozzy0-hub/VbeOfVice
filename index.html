1. functions/api/save.js

Drop into functions/api/save.js in your GitHub repo. Set ADMIN_SECRET, GITHUB_PAT, and GITHUB_REPO in Cloudflare Pages Environment Variables.

export async function onRequestPost(context) {
  const { request, env } = context;
  const secret = request.headers.get("X-Admin-Secret");

  if (!secret || secret !== env.ADMIN_SECRET) {
    return new Response(JSON.stringify({ error: "Unauthorized" }), { status: 401 });
  }

  const body = await request.json();
  const repo = env.GITHUB_REPO; // e.g. "username/vibeofvice.github.io"
  const path = "content.json";
  const apiUrl = `https://api.github.com/repos/${repo}/contents/${path}`;

  const existingReq = await fetch(apiUrl, {
    headers: {
      "Authorization": `Bearer ${env.GITHUB_PAT}`,
      "User-Agent": "VibeofVice-CMS"
    }
  });

  let sha = undefined;
  if (existingReq.ok) {
    const existingData = await existingReq.json();
    sha = existingData.sha;
  }

  const encoded = btoa(unescape(encodeURIComponent(JSON.stringify(body, null, 2))));
  const commitRes = await fetch(apiUrl, {
    method: "PUT",
    headers: {
      "Authorization": `Bearer ${env.GITHUB_PAT}`,
      "Content-Type": "application/json",
      "User-Agent": "VibeofVice-CMS"
    },
    body: JSON.stringify({
      message: "chore(cms): update VibeofVice content.json",
      content: encoded,
      sha
    })
  });

  return new Response(JSON.stringify({ ok: commitRes.ok }), {
    status: commitRes.ok ? 200 : 500,
    headers: { "Content-Type": "application/json" }
  });
}

2. n8n Merge-Safe Code Node

Ensures n8n automated runs update content.json without overwriting any card where locked_by_admin: true.

// n8n Code Node: Merge Scraped Items into VibeofVice content.json
const currentContent = $input.first().json.contentJson || {};
const existingItems = currentContent.catalog || [];
const scrapedItems = $input.first().json.newScrapedItems || [];

const lockedMap = new Map(
  existingItems.filter(i => i.locked_by_admin).map(i => [i.id, i])
);

const merged = scrapedItems.map((item, idx) => {
  if (lockedMap.has(item.id)) return lockedMap.get(item.id);
  return {
    ...item,
    source: 'n8n',
    locked_by_admin: false,
    sort_order: existingItems.length + idx + 1
  };
});

existingItems.forEach(item => {
  if ((item.locked_by_admin || item.source === 'manual') && !merged.some(m => m.id === item.id)) {
    merged.unshift(item);
  }
});

return [{
  json: {
    ...currentContent,
    catalog: merged,
    updatedAt: new Date().toISOString()
  }
}];
